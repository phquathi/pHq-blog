## acwing 3587.连通图

[3587. 连通图 - AcWing题库](https://www.acwing.com/problem/content/description/3590/)

这题是一道经典的带权并查集

如果x和y不相连，就把它们合并

最后判断最大的图的边数与n相不相等

最后就是模板即可

```c++
#include<bits/stdc++.h>
using namespace std ;

#define ll long long
#define ull unsigned long long
#define pb emplace_back
#define gtc cin.get()
#define endl (char)10

const int N = 1e3 + 10 ;
int dx[] = { -1, 0, 1, 0 };
int dy[] = { 0, -1, 0, 1 };

int p[N] , v[N];

int Find( int x )//返回X的祖宗节点+路径压缩
{
    if ( p[x] != x ) p[x] = Find(p[x]) ;
    return p[x] ;
}

int main () {
    int n , m ;
    while( cin >> n >> m ){
        for ( int i = 1 ; i <= n ; i ++ ){
            p[i] = i ;
            v[i] = 1 ;
        }
        for( int i = 1 ; i <= m ; i++ ){
            int x , y ;
            cin >> x >> y ;
            int a , b ;
            a = Find(x) ;
            b = Find(y) ;
            if (a != b){
                p[a] = b ;
                v[b] += v[a] ;
            }
        }
        if ( v[Find(1)] == n ) cout << "YES" << endl;
        else cout << "NO" << endl ;
    }
	return 0 ;
	}
```
