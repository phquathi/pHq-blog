## B.鸡兔同笼

初中数学，设共有a只鸡b只兔

列方程即可

```c++
#include<bits/stdc++.h>
using namespace std ;

#define ll long long
#define ull unsigned long long
#define pb emplace_back
#define gtc cin.get()
#define endl (char)10

const int N = 1e3 + 10 ;
int dx[] = { -1, 0, 1, 0 };
int dy[] = { 0, -1, 0, 1 };

int main () {
    int t ;
    cin >> t ;
    while ( t -- ){
        ll x , y , h , f ;
        cin >> x >> y >> h >> f ;
        cout << h - (f - x*h)/(y - x) << ' ' << (f - x*h)/(y - x) << endl ;
    }
	return 0 ;
	}
```

## C.小杜的签到

这题乍一看有点哈人，但其实看到**任意次数操作**就基本可以想到贪心

将数组排序，再把最大的与最小的元素相互作差即可

```c++
#include<bits/stdc++.h>
using namespace std ;

#define ll long long
#define ull unsigned long long
#define pb emplace_back
#define gtc cin.get()
#define endl (char)10

const int N = 2e6 + 10 ;
int dx[] = { -1, 0, 1, 0 };
int dy[] = { 0, -1, 0, 1 };
int a[N] ;

int main () {
    int n ;
    cin >> n ;
    for ( int i = 0 ; i < n ; i ++ ) cin >> a[i] ;
    sort( a , a + n );
    cout << a[0] - a[n - 1] << ' ' << a[n - 1] - a[0] << endl ;
	return 0 ;
	}
```
