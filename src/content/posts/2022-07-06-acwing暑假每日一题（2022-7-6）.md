## acwing 3625.幂次方

[3625. 幂次方 - AcWing题库](https://www.acwing.com/problem/content/3628/)

快速幂模板题，熟练模板即可

```c++
#include<bits/stdc++.h>
using namespace std ;

#define ll long long
#define ull unsigned long long
#define pb emplace_back
#define gtc cin.get()
#define endl (char)10

const int N = 1e3 + 10 ;
int dx[] = { -1, 0, 1, 0 };
int dy[] = { 0, -1, 0, 1 };

ll kuaisu( ll x , ll y , ll m){//x为底数，y为指数，m为要取模的数
    ll res = 1 ;
    while( y > 0 ){
        if ( y % 2 == 0 ){//判断指数是否为偶数
            y /= 2 ;//把指数缩小为原来的一半
            x = x * x % m ;//底数扩大为原来的平方
        }
        else {
            y -- ;//使y变为偶数
            res = res * x % m ;//此时把指数为奇数时分离出的底数的一次方收集好
            y /= 2 ;//此时指数为偶数，可以继续执行操作
            x = x * x % m ;
        }
    }
    return res ;
}

int main () {
    int a = 233333 ;
    int x , n ;
    cin >> x >> n ;
    cout << kuaisu( x , n , a ) << endl ;
   return 0 ;
   }
```

需要注意此模板并非快速幂最优解，在数据量极大的情况下还需使用位运算的方法
